  <!-- ——— JAVASCRIPT ——— -->

  <script>

    /* Utility: current year */

    document.getElementById("year").textContent = new Date().getFullYear();


    /* ——— RSS CONFIG ——— */

    // Add or remove feeds freely; the code automatically pulls the latest 5 items from each.

    const feeds = {

      india: [

        {

          url: "https://www.architecturaldigest.in/topic/architecture/rss",

          source: "Architectural Digest India",

        },

        {

          url: "https://worldarchitecture.org/feed/news/india",

          source: "World Architecture News (India)",

        },

      ],

      world: [

        {

          url: "https://www.dezeen.com/architecture/feed/",

          source: "Dezeen — Architecture",

        },

        {

          url: "https://www.archdaily.com/rss",

          source: "ArchDaily",

        },

      ],

      competitions: [

        {

          url: "https://competitions.archi/feed/",

          source: "Competitions.Archi",

        },

        {

          url: "https://archinect.com/competitions.xml",

          source: "Archinect Competitions",

        },

      ],

    };


    /* ——— FUNCTION: Fetch & render a single RSS feed ——— */

    async function fetchFeed(feed) {

      // ✱ rss2json.com offers a free API for quick prototyping (100 req/day).

      //   Feel free to replace with your own proxy or serverless function as needed.

      const api = `https://api.rss2json.com/v1/api.json?rss_url=${encodeURIComponent(

        feed.url

      )}`;


      const res = await fetch(api);

      if (!res.ok) return { source: feed.source, items: [] };

      const data = await res.json();

      return { source: feed.source, items: data.items || [] };

    }


    /* ——— FUNCTION: Render items to the DOM ——— */

    function renderItems(listEl, source, items) {

      items.slice(0, 5).forEach((item) => {

        const article = document.createElement("article");

        article.className = "border-b pb-4";

        article.innerHTML = `

          <a href="${item.link}" target="_blank" rel="noopener" class="text-lg font-medium hover:underline">

            ${item.title}

          </a>

          <p class="text-xs text-gray-500 mt-1">

            Source: ${source} • ${new Date(item.pubDate).toLocaleDateString()}

          </p>

        `;

        listEl.appendChild(article);

      });

    }


    /* ——— FUNCTION: Load a whole section ——— */

    async function loadSection(key, containerId) {

      const container = document.getElementById(containerId);

      const promises = feeds[key].map(fetchFeed);

      const results = await Promise.all(promises);

      results.forEach((result) => renderItems(container, result.source, result.items));

    }


    /* ——— INIT ——— */

    loadSection("india", "list-india");

    loadSection("world", "list-world");

    loadSection("competitions", "list-comp");
